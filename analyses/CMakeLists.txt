set(RDF_ANALYSES "" CACHE STRING "Semicolon-separated analyses to build (empty: discover all)")
file(GLOB analyses_children RELATIVE ${CMAKE_CURRENT_SOURCE_DIR} "*")
if(RDF_ANALYSES)
    set(analyses_children ${RDF_ANALYSES})
endif()
list(SORT analyses_children)
foreach(child ${analyses_children})
	if(IS_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/${child}")
		if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/${child}/CMakeLists.txt")
			set(RDF_ANALYSIS_REQUIRES "")
			include("${CMAKE_CURRENT_SOURCE_DIR}/${child}/rdf-dependencies.cmake" OPTIONAL)
			set(_missing_extensions "")
			foreach(_requirement IN LISTS RDF_ANALYSIS_REQUIRES)
				string(TOLOWER "${_requirement}" _requirement)
				if(NOT _requirement IN_LIST RDF_ENABLED_EXTENSIONS)
					list(APPEND _missing_extensions "${_requirement}")
				endif()
			endforeach()
			if(_missing_extensions)
				if(RDF_ANALYSES)
					message(FATAL_ERROR "Analysis ${child} requires missing or disabled extensions: ${_missing_extensions}. Download their checkouts and enable their extension-owned options.")
				endif()
				message(STATUS "Skipping analysis ${child}: extensions unavailable: ${_missing_extensions}")
				continue()
			endif()
			add_subdirectory(${child})
			# Only these vjet units showed a repeatable whole-job CPU benefit
			# from inline input tables over the framework's packed reader.
			if(RDF_ROOT_LOCALITY_AVAILABLE AND child STREQUAL "vjet-njetpt" AND TARGET vjet_njetpt)
				set_property(SOURCE
					"${CMAKE_CURRENT_SOURCE_DIR}/${child}/src/analysis.cc"
					"${CMAKE_CURRENT_SOURCE_DIR}/${child}/src/BackgroundStudy.cc"
					TARGET_DIRECTORY vjet_njetpt APPEND PROPERTY
					INCLUDE_DIRECTORIES "${RDF_ROOT_INLINE_INCLUDE}")
			endif()
		elseif(RDF_ANALYSES)
			message(FATAL_ERROR "Requested analysis ${child} has no CMakeLists.txt")
		endif()
	elseif(RDF_ANALYSES)
		message(FATAL_ERROR "Requested analysis ${child} does not exist")
	endif()
endforeach()
